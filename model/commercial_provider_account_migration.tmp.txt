const commercialProviderAccountTableDDL = `
CREATE TABLE public.commercial_provider_accounts (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  channel_id integer NOT NULL,
  provider_code varchar(64) NOT NULL,
  account_label varchar(128) NOT NULL,
  max_concurrency integer NOT NULL DEFAULT 6,
  route_weight integer NOT NULL DEFAULT 100,
  spend_cap_microusd bigint NULL,
  config_version bigint NOT NULL DEFAULT 1,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT fk_commercial_provider_accounts_channel
    FOREIGN KEY (channel_id) REFERENCES public.channels(id) ON DELETE RESTRICT,
  CONSTRAINT ck_commercial_provider_accounts_provider_code
    CHECK (provider_code ~ '^[a-z0-9][a-z0-9._-]{0,63}$'),
  CONSTRAINT ck_commercial_provider_accounts_account_label
    CHECK (length(account_label) BETWEEN 1 AND 128
      AND position(E'\r' in account_label) = 0
      AND position(E'\n' in account_label) = 0),
  CONSTRAINT ck_commercial_provider_accounts_bounds
    CHECK (max_concurrency BETWEEN 1 AND 6
      AND route_weight BETWEEN 1 AND 1000
      AND (spend_cap_microusd IS NULL OR spend_cap_microusd BETWEEN 0 AND 1000000000000000)
      AND config_version BETWEEN 1 AND 1000000000)
)`

const commercialProviderAccountGuardFunctionDDL = `
CREATE OR REPLACE FUNCTION public.commercial_provider_account_channel_guard_fn()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
  commercial_channel boolean;
  channel_key text;
  secret_ref text;
  channel_config_version bigint;
BEGIN
  SELECT c.commercial_channel, c.key, c.secret_ref, c.config_version
    INTO commercial_channel, channel_key, secret_ref, channel_config_version
    FROM public.channels AS c
   WHERE c.id = NEW.channel_id;
  IF NOT FOUND OR commercial_channel IS NOT TRUE THEN
    RAISE EXCEPTION 'pool channel is not commercial';
  END IF;
  IF coalesce(channel_key, '') <> '' THEN
    RAISE EXCEPTION 'pool channel contains raw credential material';
  END IF;
  IF secret_ref IS NULL OR secret_ref !~ '^env://[A-Z_][A-Z0-9_]*$' THEN
    RAISE EXCEPTION 'pool channel secret reference is invalid';
  END IF;
  IF coalesce(channel_config_version, 0) <= 0 THEN
    RAISE EXCEPTION 'pool channel configuration is inactive';
  END IF;
  RETURN NEW;
END
$$`

const commercialProviderAccountUpdatedFunctionDDL = `

  RETURN NEW;
END
$$`

func commercialProviderAccountEnsureTrigger(tx *gorm.DB, trigger, table, functionDDL, triggerDDL string) error {
	functionName := commercialProviderAccountGuardFn
	if strings.Contains(functionDDL, commercialProviderAccountUpdatedFn) {
		functionName = commercialProviderAccountUpdatedFn
	}
	regproc := "public." + functionName + "()"
	var functionOID *string
	if err := tx.Raw("SELECT to_regprocedure(?)::oid::text", regproc).Scan(&functionOID).Error; err != nil {
		return err
	}
	if functionOID != nil && strings.TrimSpace(*functionOID) != "" {
		var comment *string
		if err := tx.Raw("SELECT obj_description(?::oid, 'pg_proc')", *functionOID).Scan(&comment).Error; err != nil {
			return err
		}
		if comment == nil || *comment != commercialProviderAccountOwner {
			return fmt.Errorf("commercial provider account function %s is not migration-owned", functionName)
		}
	}
	if err := tx.Exec(functionDDL).Error; err != nil {
		return err
	}
	if err := tx.Exec("COMMENT ON FUNCTION " + regproc + " IS " + commercialProviderAccountSQLLiteral(commercialProviderAccountOwner)).Error; err != nil {
		return err
	}
	var present bool
	if err := tx.Raw("SELECT EXISTS (SELECT 1 FROM pg_trigger WHERE tgname = ? AND tgrelid = 'public."+table+"'::regclass AND NOT tgisinternal)", trigger).Scan(&present).Error; err != nil {
		return err
	}
	if !present {
		if err := tx.Exec(triggerDDL).Error; err != nil {
			return err
		}
		if err := tx.Exec("COMMENT ON TRIGGER " + trigger + " ON public." + table + " IS " + commercialProviderAccountSQLLiteral(commercialProviderAccountOwner)).Error; err != nil {
			return err
		}
		return nil
	}
	var comment *string
	if err := tx.Raw("SELECT obj_description(t.oid, 'pg_trigger') FROM pg_trigger AS t WHERE t.tgname = ? AND t.tgrelid = 'public."+table+"'::regclass AND NOT t.tgisinternal", trigger).Scan(&comment).Error; err != nil {
		return err
	}
	if comment == nil || *comment != commercialProviderAccountOwner {
		return fmt.Errorf("commercial provider account trigger %s is not migration-owned", trigger)
	}
	return nil
}

func commercialProviderAccountCatalogOID(tx *gorm.DB, object commercialProviderAccountLedgerObject) (string, error) {
	var oid string
	var err error
	switch object.ObjectType {
	case "table", "index":
		err = tx.Raw("SELECT COALESCE(to_regclass(?)::oid::text, '')", object.SchemaName+"."+object.ObjectName).Scan(&oid).Error
	case "function":
		err = tx.Raw("SELECT COALESCE(to_regprocedure(?)::oid::text, '')", object.SchemaName+"."+object.ObjectName).Scan(&oid).Error
	case "trigger":
		err = tx.Raw("SELECT COALESCE(t.oid::text, '') FROM pg_trigger AS t JOIN pg_class AS c ON c.oid = t.tgrelid JOIN pg_namespace AS n ON n.oid = c.relnamespace WHERE n.nspname = ? AND c.relname = ? AND t.tgname = ? AND NOT t.tgisinternal", object.SchemaName, commercialProviderAccountTable, object.ObjectName).Scan(&oid).Error
	default:
		return "", fmt.Errorf("unsupported provider-account ledger object type %s", object.ObjectType)
	}
	if err != nil {
		return "", err
	}
	if strings.TrimSpace(oid) == "" {
		return "", fmt.Errorf("commercial provider account object %s.%s is missing", object.SchemaName, object.ObjectName)
	}
	return oid, nil
}

		}

func commercialProviderAccountVerifyLedger(tx *gorm.DB) error {
	var owner *string
	if err := tx.Raw("SELECT obj_description(('public.' || ?)::regclass, 'pg_class')", commercialProviderAccountLedger).Scan(&owner).Error; err != nil {
		return err
	}
	if owner == nil || *owner != commercialProviderAccountOwner {
		return errors.New("commercial provider account migration ledger is not migration-owned")
	}
	for _, object := range commercialProviderAccountLedgerObjects() {
		var storedHash, state string
		var validated bool
		var objectOID *string
		row := tx.Raw("SELECT definition_hash, state, validated, object_oid::text FROM public."+commercialProviderAccountLedger+" WHERE migration_id = ? AND object_type = ? AND schema_name = ? AND object_name = ?", commercialProviderAccountMigrationID, object.ObjectType, object.SchemaName, object.ObjectName).Row()
		if err := row.Scan(&storedHash, &state, &validated, &objectOID); err != nil {
			return fmt.Errorf("commercial provider account ledger missing %s.%s: %w", object.SchemaName, object.ObjectName, err)
		}
		if storedHash != object.Hash || state != "complete" || !validated || objectOID == nil || strings.TrimSpace(*objectOID) == "" {
			return fmt.Errorf("commercial provider account ledger mismatch for %s.%s", object.SchemaName, object.ObjectName)
		}
		catalogOID, err := commercialProviderAccountCatalogOID(tx, object)
		if err != nil || catalogOID != *objectOID {
			return fmt.Errorf("commercial provider account catalog ownership mismatch for %s.%s", object.SchemaName, object.ObjectName)
		}
	}
	return nil
}
func migrateCommercialProviderAccountFoundation() error {
	if DB == nil || !common.UsingMainDatabase(common.DatabaseTypePostgreSQL) {
		return nil
	}
	return DB.Transaction(func(tx *gorm.DB) error {
		if err := tx.Exec("SELECT pg_advisory_xact_lock(CAST(? AS integer), CAST(? AS integer))", commercialProviderAccountLockClass, commercialProviderAccountLockObject).Error; err != nil {
			return fmt.Errorf("commercial provider account migration lock: %w", err)
		}
		for _, column := range []string{"commercial_channel", "secret_ref", "auth_type", "config_version"} {
			var present bool
			if err := tx.Raw("SELECT EXISTS (SELECT 1 FROM information_schema.columns WHERE table_schema = 'public' AND table_name = 'channels' AND column_name = ?)", column).Scan(&present).Error; err != nil {
				return err
			}
			if !present {
				return fmt.Errorf("commercial provider account migration requires channels.%s", column)
			}
		}
		if err := commercialProviderAccountEnsureOwnedTable(tx, commercialProviderAccountLedger, commercialProviderAccountLedgerDDL, commercialProviderAccountOwner); err != nil {
			return err
		}
		if err := commercialProviderAccountEnsureOwnedTable(tx, commercialProviderAccountTable, commercialProviderAccountTableDDL, commercialProviderAccountOwner); err != nil {
			return err
		}
		if err := commercialProviderAccountEnsureIndex(tx, "ux_commercial_provider_accounts_channel_id", "CREATE UNIQUE INDEX ux_commercial_provider_accounts_channel_id ON public.commercial_provider_accounts (channel_id)"); err != nil {
			return err
		}
		if err := commercialProviderAccountEnsureIndex(tx, "ix_commercial_provider_accounts_provider_code", "CREATE INDEX ix_commercial_provider_accounts_provider_code ON public.commercial_provider_accounts (provider_code)"); err != nil {
			return err
		}
		if err := commercialProviderAccountEnsureIndex(tx, "ix_commercial_provider_accounts_config_version", "CREATE INDEX ix_commercial_provider_accounts_config_version ON public.commercial_provider_accounts (config_version)"); err != nil {
			return err
		}
		if err := commercialProviderAccountEnsureTrigger(tx, commercialProviderAccountGuardTrigger, commercialProviderAccountTable, commercialProviderAccountGuardFunctionDDL, "CREATE TRIGGER "+commercialProviderAccountGuardTrigger+" BEFORE INSERT OR UPDATE ON public."+commercialProviderAccountTable+" FOR EACH ROW EXECUTE FUNCTION public."+commercialProviderAccountGuardFn+"()"); err != nil {
			return err
		}
		if err := commercialProviderAccountEnsureTrigger(tx, commercialProviderAccountUpdatedTrigger, commercialProviderAccountTable, commercialProviderAccountUpdatedFunctionDDL, "CREATE TRIGGER "+commercialProviderAccountUpdatedTrigger+" BEFORE UPDATE ON public."+commercialProviderAccountTable+" FOR EACH ROW EXECUTE FUNCTION public."+commercialProviderAccountUpdatedFn+"()"); err != nil {
			return err
		}
		ready, err := commercialProviderAccountSchemaReady(tx)
		if err != nil {
			return err
		}
		if !ready {
			return ErrCommercialProviderAccountSchemaUnavailable
		}
		if err := commercialProviderAccountRecordLedger(tx); err != nil {
			return err
		}
		return commercialProviderAccountVerifyLedger(tx)
	})
}
